cmake_minimum_required(VERSION 3.16)
project(aster_script_host C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(aster-script-host host.c "${QJS_SOURCE}/quickjs.c" "${QJS_SOURCE}/libregexp.c" "${QJS_SOURCE}/libunicode.c" "${QJS_SOURCE}/dtoa.c")
target_include_directories(aster-script-host PRIVATE "${QJS_SOURCE}")
target_compile_definitions(aster-script-host PRIVATE _GNU_SOURCE QUICKJS_NG_BUILD)
if(WIN32)
  target_compile_definitions(aster-script-host PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0602 _CRT_SECURE_NO_WARNINGS)
else()
  target_link_libraries(aster-script-host PRIVATE m pthread dl)
endif()
